/* C Stack Data Structure & Word Reversal Logic */
#define MAX_SIZE 1024

typedef struct {
    int top;
    char items[MAX_SIZE];
} Stack;

void initStack(Stack *s) { s->top = -1; }
int  isFull(Stack *s)    { return s->top >= MAX_SIZE - 1; }
int  isEmpty(Stack *s)   { return s->top < 0; }

int push(Stack *s, char ch) {
    if (isFull(s)) return -1; // Overflow
    s->items[++(s->top)] = ch;
    return s->top;
}

char pop(Stack *s) {
    if (isEmpty(s)) return '\0'; // Stack Underflow
    return s->items[(s->top)--];
}

char* reverseWord(const char *in, char *out) {
    Stack s;
    initStack(&s);
    // Step 2: Push characters one by one
    for (int i = 0; in[i]; i++) push(&s, in[i]);
    // Step 3: Pop elements until stack underflow
    int j = 0;
    while (!isEmpty(&s)) out[j++] = pop(&s);
    out[j] = '\0';
    return out; // Step 4: Reversed string
}
